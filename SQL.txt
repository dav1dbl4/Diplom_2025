CREATE OR REPLACE FUNCTION format_phone_number()
RETURNS TRIGGER AS $$
BEGIN
    NEW."Номер телефона" := regexp_replace(NEW."Номер телефона", '\D', '', 'g');
    
    IF length(NEW."Номер телефона") = 11 THEN
        NEW."Номер телефона" := '8 (' ||
            SUBSTRING(NEW."Номер телефона", 2, 3) || ') ' ||
            SUBSTRING(NEW."Номер телефона", 5, 3) || '-' ||
            SUBSTRING(NEW."Номер телефона", 8, 2) || '-' ||
            SUBSTRING(NEW."Номер телефона", 10, 2);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER trigger_format_phone
BEFORE INSERT OR UPDATE ON buyers
FOR EACH ROW
EXECUTE FUNCTION format_phone_number();

create table remains("Код товара" int, "Товар" varchar, "Единица измерения(кг/шт)" quantity_type, "Остаток" int);

CREATE OR REPLACE FUNCTION fill_stock_remains()
RETURNS TRIGGER AS $$
BEGIN
    SELECT "Товар", "Единица измерения(кг/шт)" INTO NEW."Товар", NEW."Единица измерения(кг/шт)" FROM stock WHERE "Код товара" = NEW."Код товара" LIMIT 1;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER auto_fill_stock_remains
BEFORE INSERT OR UPDATE ON remains
FOR EACH ROW
EXECUTE FUNCTION fill_stock_remains();


CREATE OR REPLACE FUNCTION calculate_remains()
RETURNS void AS $$
BEGIN
    UPDATE remains r
    SET "Остаток" = COALESCE((
        SELECT SUM(m."Количество" - m."Распродано/Списано")
        FROM makings m
        WHERE m."Код товара" = r."Код товара"
        GROUP BY m."Код товара"
    ), 0);
    
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION sync_remains()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO remains ("Код товара", "Товар")
    VALUES (NEW."Код товара", NEW."Товар");
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER after_insert_stock
AFTER INSERT ON stock
FOR EACH ROW
EXECUTE FUNCTION sync_remains();

create table makings("Код изготовления" serial primary key, "Код товара" int, "Товар" varchar, "Дата изготовления" date, "Единица измерения(кг/шт)" quantity_type, "Количество" int, "Распродано/Списано" int);


CREATE OR REPLACE FUNCTION decrease_makings(product_code int, amount int)
RETURNS VOID AS $$
declare rec RECORD;
BEGIN
    WHILE amount > 0 LOOP
	select * into rec from makings where "Код товара"=product_code and "Количество" <> "Распродано/Списано" ORDER BY "Дата изготовления" ASC
        LIMIT 1;
	IF rec."Количество"-rec."Распродано/Списано" >= amount THEN
		UPDATE makings
		SET "Распродано/Списано" = "Распродано/Списано" + amount
                WHERE "Код товара" = product_code 
                  AND "Дата изготовления" = rec."Дата изготовления";
                amount := 0;
	ELSE
		amount := amount - rec."Количество" + rec."Распродано/Списано";
		UPDATE makings
                SET "Распродано/Списано" = "Количество"  -- Устанавливаем количество в 0;
                WHERE "Код товара" = product_code 
                  AND "Дата изготовления" = rec."Дата изготовления";
        END IF;
    END LOOP;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION is_expired(production_code INT)
RETURNS BOOLEAN AS $$
DECLARE
    manufacturing_date DATE;
    shelf_life INT;
BEGIN
    SELECT m."Дата изготовления", s."Срок годности(дни)"
    INTO manufacturing_date, shelf_life
    FROM makings m
    JOIN stock s using("Код товара")
    WHERE m."Код изготовления" = production_code;

    RETURN (current_date - manufacturing_date > shelf_life);
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION find_expired()
RETURNS table ("Код товара" integer) AS $$
    select m."Код товара" from makings m join stock s using("Код товара") where current_date - m."Дата изготовления" > s."Срок годности(дни)" and m."Количество"<> m."Распродано/Списано"; 
$$ LANGUAGE sql;


CREATE OR REPLACE FUNCTION find_expired()
RETURNS table ("Код товара" integer) AS $$
BEGIN
    RETURN QUERY 
    SELECT m."Код товара"
    FROM makings m
    JOIN stock s USING("Код товара")
    WHERE current_date - m."Дата изготовления" > s."Срок годности(дни)" 
      AND m."Количество" <> m."Распродано/Списано";

    IF NOT FOUND THEN
        RETURN QUERY SELECT 0;
    END IF;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION spisat(production_code INT)
RETURNS void AS $$
DECLARE
    code int;
    tovar varchar;
    manufacturing_date DATE;
    izmerenie quantity_type;
    spisano INT;
BEGIN
    Select m."Код товара", m."Товар", st."Единица измерения(кг/шт)", m."Количество" - m."Распродано/Списано" into code, tovar, izmerenie, spisano
	from makings m
		join stock st using("Код товара")
		where m."Код изготовления" = production_code;
    INSERT INTO bad values(code, tovar, current_date, izmerenie, spisano);
    UPDATE makings SET "Распродано/Списано" = "Количество" WHERE "Код изготовления" = production_code; 
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION fill_stock_makings()
RETURNS TRIGGER AS $$
BEGIN
    SELECT "Товар", "Единица измерения(кг/шт)" INTO NEW."Товар", NEW."Единица измерения(кг/шт)" FROM stock WHERE "Код товара" = NEW."Код товара" LIMIT 1;
    IF NEW."Распродано/Списано" IS NULL THEN
        NEW."Распродано/Списано" := 0; 
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER auto_fill_stock_makings
BEFORE INSERT OR UPDATE ON makings
FOR EACH ROW
EXECUTE FUNCTION fill_stock_makings();

CREATE OR REPLACE FUNCTION calculate_revenue()
RETURNS numeric(10,2) AS $$
DECLARE
    temp int := SUM(b."Количество" * st."Себестоимость(тенге)") from bad b join stock st using("Код товара");
begin
    return (SUM(s."Количество" * (st."Цена продажи(тенге)" - st."Себестоимость(тенге)")) - temp)
    FROM sales s
    JOIN stock st using("Код товара");
end;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION calculate_revenue_by_period(timestamp, timestamp)
RETURNS numeric(10,2) AS $$
DECLARE
    temp int := SUM(b."Количество" * st."Себестоимость(тенге)") from bad b join stock st using("Код товара") where b."Дата списания" between $1 and $2;
begin
    return (SUM(s."Количество" * (st."Цена продажи(тенге)" - st."Себестоимость(тенге)")) - temp)
    FROM sales s
    JOIN stock st using("Код товара")
	where s."Дата продажи" between $1 and $2;
end;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION calculate_bestseller()
RETURNS table("Товар" varchar, "Количество продаж" integer)
as $$
SELECT st."Товар", sum(s."Количество") as "Количество продаж"
    FROM sales s
    JOIN stock st using("Код товара")
group by st."Товар"
order by "Количество продаж" desc
limit 1; 
$$ LANGUAGE sql;


CREATE OR REPLACE FUNCTION calculate_bestseller_by_period(timestamp, timestamp)
RETURNS table("Товар" varchar, "Количество продаж" integer)
as $$
SELECT st."Товар", sum(s."Количество") as "Количество продаж"
    FROM sales s
    JOIN stock st using("Код товара")
	where s."Дата продажи" between $1 and $2
group by st."Товар"
order by "Количество продаж" desc
limit 1; 
$$ LANGUAGE sql;

AS $$
declare
    temp numeric(10,2);
begin
    select SUM(b."Количество" * st."Себестоимость(тенге)") into temp from bad b join stock st using("Код товара");
    return query
SELECT SUM(s."Количество" * (st."Цена продажи(тенге)")) as "Продажи", SUM(s."Количество" * (st."Себестоимость(тенге)")) + temp as "Расходы"
    FROM sales s
    JOIN stock st using("Код товара");
end;
$$ LANGUAGE plpgsql;
